        // 增減數量的便捷按鈕腳本

        function updateQty(inputId, change) {

            const input = document.getElementById(inputId);

            if (!input) return;

            let currentVal = parseInt(input.value) || 0;

            let newVal = currentVal + change;

            if (newVal < 0) newVal = 0;

            input.value = newVal;

            calculateTotal(); // 觸發計算

        }


        // 自動計算總金額與運費腳本

        function calculateTotal() {

            let total = 0;

            const items = document.querySelectorAll('.product-qty');

            

            // 計算小計

            items.forEach(item => {

                const qty = parseInt(item.value) || 0;

                const price = parseInt(item.dataset.price) || 0;

                total += (qty * price);

            });

            

            // 計算運費 (未滿 1299 收 200)

            let shipping = 0;

            if (total > 0 && total < 1299) {

                shipping = 200;

            }

            

            // 滿額贈品提示 (滿 1800)

            const promoEl = document.getElementById('promo-message');

            if (promoEl) {

                if (total >= 1800) {

                    promoEl.classList.remove('hidden');

                } else {

                    promoEl.classList.add('hidden');

                }

            }

            

            // 總計

            const finalTotal = total + shipping;

            

            // 更新 UI 顯示

            const displaySubtotal = document.getElementById('display-subtotal');

            const displayShipping = document.getElementById('display-shipping');

            const displayTotal = document.getElementById('display-total');

            const inputTotal = document.getElementById('input-total');


            if (displaySubtotal) displaySubtotal.innerText = total.toLocaleString();

            if (displayShipping) displayShipping.innerText = shipping.toLocaleString();

            if (displayTotal) displayTotal.innerText = finalTotal.toLocaleString();

            

            // 寫入傳送至 Google 表單的隱藏 Input

            if (inputTotal) {

                inputTotal.value = finalTotal > 0 ? finalTotal : '';

                

                // 同時觸發 HTML5 Required 驗證重置

                if(finalTotal > 0) {

                    inputTotal.setCustomValidity("");

                }

            }

        }


        // 表單送出相關處理

        let isSubmitted = false;


        function handleFormSubmit(e) {

            const inputTotal = document.getElementById('input-total');

            const finalTotal = inputTotal ? (parseInt(inputTotal.value) || 0) : 0;

            if (finalTotal === 0) {

                alert("您尚未選擇任何商品！");

                e.preventDefault();

                return false;

            }


            const btn = document.getElementById('submitBtn');

            if (btn) {

                btn.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i> 訂單送出中...';

                btn.disabled = true;

            }

            isSubmitted = true;

            

            // 表單將會被 target 到隱藏的 iframe

        }
